=============================
1. install git
==============================
sudo apt update && sudo apt install git -y

==============================
2. install mise & configure
==============================
curl https://mise.run | sh

eval "$(~/.local/bin/mise activate zsh)"

mkdir -p ~/.config/mise && touch ~/.config/mise/config.toml

echo $'<copy raw toml from dotfiles repo>' >> .config/mise/config.toml

mise settings set npm.package_manager pnpm


==============================
3. edit ~/.bashrc
==============================
hx ~/.bashrc -> update from dotfiles repo

==============================
4. copy configs from host
==============================

mkdir -p ~/.config/jj

orb push -m mustafar ~/.config/jj/config.toml  ~/.config/jj/config.toml
orb push -m mustafar ~/.gitconfig  ~/.gitconfig
orb push -m mustafar ~/.gitallowedsigners  ~/.gitallowedsigners
