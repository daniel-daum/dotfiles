=============================
1. add standard packages
==============================
sudo apt update
sudo apt modernize-sources
sudo apt install -y nano vim-tiny less software-properties-common ca-certificates curl gnupg wget xz-utils git
sudo apt update

==============================
2. install mise & configure
==============================
curl https://mise.run | sh
echo 'eval "$(~/.local/bin/mise activate bash)"' >> ~/.bashrc

==============================
3. install global mise dependencies
==============================
mise use -g bat delta difftastic eza helix jj jq ripgrep

==============================
4. edit ~/.bashrc
==============================
# mise
eval "$(~/.local/bin/mise activate bash)"

# go via mise
export PATH="$HOME/go/bin:$PATH"

# eza ls replacement
alias ll="eza -al --group-directories-first --icons --git --color=auto --long --header --classify"

# bat cat replacement
alias cat='bat'
