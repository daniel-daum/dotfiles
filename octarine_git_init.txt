==============================
1. update ssh config
==============================
Host git.ts.danieldaum.net-automatic
    HostName git.ts.danieldaum.net
    Port 2222
    IdentityFile ~/.ssh/automatic_secretive.pub
    IdentitiesOnly yes

==============================
2. copy key to path
==============================
cp /<ssh-key-path> ~/.ssh/automatic_secretive.pub

==============================
3. ensure repo url uses ssh alias
==============================
git remote set-url origin git@git.ts.danieldaum.net-automatic:your/repo.git

==============================
4. turn off gpg signgin
==============================
git config --local commit.gpgsign false

==============================
5. init via push
==============================
git push -u origin main
