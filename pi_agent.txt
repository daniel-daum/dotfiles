==============================
1. install drop
==============================
sudo apt install -y passt
ARCH=$(uname -m | sed 's/x86_64/amd64/; s/aarch64/arm64/')
curl -o /tmp/drop -L "https://github.com/wrr/drop/releases/latest/download/drop-linux-$ARCH"
mkdir -p ~/.local/bin
install /tmp/drop ~/.local/bin/
rm /tmp/drop
source ~/.profile

# check (should print root)
unshare --user --map-root-user whoami

==============================
2. install pi + node via mise
==============================
mise use -g pi@latest
mise use -g node@lts

==============================
3. install extensions (outside drop)
==============================
pi install npm:pi-lens
pi install npm:@juicesharp/rpiv-todo
pi install npm:@juicesharp/rpiv-ask-user-question
pi install npm:pi-subagents
pi install npm:context-mode

==============================
